3.2 If The Else Statement

# c++ way
x <- 10
if (x > 0) y = sqrt(x) else y = -sqrt(x)
y
[1] 3.162278
# rust or lisp or algol way
y = if (x > 0) sqrt(x) else -sqrt(x)
y
[1] 3.162278

3.2.1 Combining Logical Conditions

&& and || operate evaluate just enough of their arguments

TRUE && print(TRUE)
[1] TRUE
[1] TRUE
## both arguments evaluated
TRUE || print(TRUE)
[1] TRUE
## only evaluate the first, because it is enough to know the whole results

threshold <- 100
n <- 0

while (s < threshold) {
    n = n + 1

# gucci gang
dfs = function(remain, temp_str, lst){
    if (remain == 0){
        lst <- append(lst, list(temp_str))
        return(lst)
    }

    if (remain >= 1) {
        next_str <- if (temp_str == "") "gang" else
            paste(temp_str, "gang", sep = "")
        lst <- dfs(remain - 1, next_str, lst);
    }

    if (remain >= 2) {
        next_str <- if (temp_str == "") "gucci" else
            paste(temp_str, "gucci", sep = "");
        lst <- dfs(remain - 2, next_str, lst)
    }

    lst
}

gucci.gang = function(x){
    if (mode(x) != 'numeric') {
        print("error");
        return(NULL);
    }

    lst <- list();
    y <- '';
    lst <- dfs(x, y, lst);
    lst <- as.character(lst);
    lst
}

gucci.gang(3)
[1] "gangganggang" "ganggucci"    "guccigang"

repeat loop with break

threshold <- 100
n <- 0
s <- 0
repeat {
    if (s > threshold) break
    n = n + 1
    s = s + n
}
c(n, s)
[1]  14 105
