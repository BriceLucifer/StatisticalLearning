# gucci gang
dfs = function(remain, temp_str, lst){
    if (remain == 0){
        lst <- append(lst, list(temp_str))
        return(lst)
    }

    if (remain >= 1) {
        next_str <- if (temp_str == "") "gang" else
            paste(temp_str, "gang", sep = "")
        lst <- dfs(remain - 1, next_str, lst);
    }

    if (remain >= 2) {
        next_str <- if (temp_str == "") "gucci" else
            paste(temp_str, "gucci", sep = "");
        lst <- dfs(remain - 2, next_str, lst)
    }

    lst
}

if expr has a numeric value equal to n, then the nth of the code blocks is executed.

iterative procedure for computing square roots for positive numbers

square.root = function(x, iterator = 10, start = 1){
    r.new = start
    r.old = 0
    while (iterator != 0){
        if (all(r.new == r.old)) break
        r.old = r.new
        r.new = (r.new + x / r.new) / 2
        iterator = iterator - 1
    }
    r.new
}
